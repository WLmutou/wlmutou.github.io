1. 找数据

emmmmm,还以为要正则匹配页面呢。原来有api啦。
那么，直接简单curl请求是否以?当然是不行，得加头

func DoHttpGet(url string) ([]byte, error) {
    client := &http.Client{}
    reqest, err := http.NewRequest("GET", url, nil )
    if err != nil {
        fmt.Println("Fatal error ", err.Error())
        os.Exit(0)
    }
    reqest.Header.Add("Accept", "application/json, text/javascript, */*; q=0.01")
    //reqest.Header.Add("Accept-Encoding", "gzip,deflate,sdch")
    reqest.Header.Add("Accept-Language", "zh-CN,zh;q=0.9")
    reqest.Header.Add("Connection", "keep-alive")
    reqest.Header.Add("Content-Type", "application/x-www-form-urlencoded")

    reqest.Header.Add("Cookie", "Cookie:bid=8k03f9NA7xk; ap_v=0,6.0; __utmt=1; __utma=30149280.1067919873.1550371826.1550371826.1550371826.1; __utmb=30149280.1.10.1550371826; __utmc=30149280; __utmz=30149280.1550371826.1.1.utmcsr=(direct)|utmccn=(direct)|utmcmd=(none)")
    reqest.Header.Add("Host", "m.douban.com")
    reqest.Header.Add("Origin", "https://www.douban.com")
    reqest.Header.Add("Referer", "https://www.douban.com/gallery/topic/51644/?sort=new&guest_only=0")
    reqest.Header.Add("User-Agent", "Mozilla/5.0(Macintosh;IntelMacOSX10_7_0)AppleWebKit/535.11(KHTML,likeGecko)Chrome/17.0.963.56Safari/535.11")

    response, err := client.Do(reqest)
    defer response.Body.Close()

    if err != nil {
        return nil, err
    }
    buf, err := ioutil.ReadAll(response.Body)
    if err != nil {
        logrus.Errorln(err.Error())
    }

    return buf, err
}

ok，插入数据库,再页面显示应该就好了.

2.图片

恩，文字确实可以了，但是，image，却是不行.
拿到的是douban的域名链接地址,很显然，豆瓣并没有 access 跨域访问.
怎么办?
只能把图片download回来了.

来，go下载图片，这个还搞了一阵(话说，以前还特喜欢python搞这事，现在…)

func getImage(text string) {
    rcg := regexp.MustCompile(`<img src=(.*?)>`)
    urls := rcg.FindAllString(text, -1)
    for _, url := range urls {
        url = strings.Replace(url, "\"/>", "", -1)
        url = strings.Replace(url, "<img src=\"", "", -1)
        fmt.Println(url)
        downloadImg(url)
    }
}

func downloadImg(url string)  {
    if url == ""{
        return
    }

    imgPathList := strings.Split(url, "doubanio.com/")
    if len(imgPathList) <= 1{
        return
    }
    imgPath := imgPathList[1]

    localPathList := strings.Split(imgPath, "/")
    localPath := strings.Join(localPathList[:len(localPathList) - 1], "/")
    localStaticPath := strings.Trim(static, "/") + "/" + localPath

    check.CheckAndCreateDirectory(localStaticPath)

    staticImgPath := strings.Trim(static, "/") + "/" + imgPath
    if check.FileExists(staticImgPath) {
        return
    }
    resp, err := requests.DoHttpGet(url)
    if err != nil {
        logrus.Errorln(err)
        return
    }
    out, err := os.Create(staticImgPath)
    defer out.Close()
    _, err = io.Copy(out, bytes.NewReader(resp))
    time.Sleep(time.Second * 2)
}

3.解决部署遇到问题

txCloud申请的域名，没有实名认证也可以访问(caddy),aliyun却还没有申请下来(吗也，身份证到期了)
很好，tx云可以访问页面.但是，爬取的时候，却发现，txCloud上无法爬取,请求错误.而,同样的代码在本地和aliyun确实ok.
可能，douban把txcloud的ip封了.当然,也有可能txCloud我有地方没有配置好….

咋整嘞?
因为txCloud和aliyun都在吃灰,要不从aliyun下载后再拷贝过txCloud,ok就先这样玩了.
脚本搞定(导出用python啦)


tx_cloud_ip="xx.xx.xx.xx"
remote_static="/home/hhly/gopath/src/blind-date/"
local_path="/home/hhly/gopath/src/blind-date"

function start() {
    echo "dump data ..."
    dump_mysql_json.py

    echo "tar static ..."
    cd $local_path && tar -zcf /tmp/$static_name   ./static 

    echo "scp ..."

    ssh hhly@$tx_cloud_ip  "cd  /home/hhly/gopath/src/blind-date/ && rmpy.py  static.tar.gz " 

    scp -P 10089 /tmp/$sql_name  hhly@$tx_cloud_ip:/tmp/
    scp -P 10089 /tmp/$static_name  hhly@$tx_cloud_ip:/home/hhly/gopath/src/blind-date/

    cd  /tmp/ && rmpy.py $static_name
    cd  /tmp/ && rmpy.py $sql_name

    echo "ssh command ..." 
    ssh hhly@$tx_cloud_ip"cd  /home/hhly/gopath/src/blind-date/ && tar -zxf ./static.tar.gz " 
    ssh hhly@$tx_cloud_ip "cd  /home/hhly/gopath/src/blind-date && ./loadjson "
}


while true;do 
    start
    echo "sleep 300..."
    sleep 300
done

以上需要注意的是cd切换了目录，在loop回来之后，最上面的tar要cd切换到那个目录去,否则找不到文件夹呀(这坑了)
好了，现在终于，稳定运行.
