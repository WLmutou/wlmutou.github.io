#!/usr/bin/env python3

### weibo_get.py ###

from urllib.request import quote
import requests 
import codecs
from re_weibo import ParseHTML
from tkinter import scrolledtext

search_weibo_url = "https://s.weibo.com/weibo?q=%s&wvr=6&b=1&Refer=SWeibo_box"

headers = {'Accept': 'text/html,application/xhtml+xml,application/xml;q=0.9,image/webp,image/apng,*/*;q=0.8,application/signed-exchange;v=b3;q=0.9',
            'Accept-Encoding':'gzip, deflate, br',
            'Accept-Language': 'zh-CN,zh;q=0.9',
            'Connection':'Keep-Alive',
            'Host':'s.weibo.com',
            'Referer': '', # https://s.weibo.com/weibo?q=%s&wvr=6&b=1&Refer=SWeibo_box
            'Sec-Fetch-Dest': 'document',
            'Sec-Fetch-Mode': 'navigate',
            'Sec-Fetch-Site': 'same-origin',
            'Sec-Fetch-User': '?1',
            'Upgrade-Insecure-Requests': '1',
            'User-Agent':'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/52.0.2743.116 Safari/537.36 Edge/15.15063'}

def search_weibo(key:str) -> ([], str):
    req_url = search_weibo_url % ( quote(key)) 
    headers['Referer'] = req_url
    try:
        resp = requests.get(req_url,  headers=headers, timeout=5)
        all_item = ParseHTML(resp.text)
        return all_item, ""
    except Exception as e:
        print (e)
        return [], e 
    return [], ""

使用bs4解析

把获取到的结果解析，拿到上面的要的值

#!/usr/bin/env python3
# encoding:utf-8

### re_weibo.py ###

import re 
from bs4 import element
from bs4 import BeautifulSoup #导入库
from datetime import datetime
from datetime import timedelta


################## 需要的字段 #####################
# 1. 博主id
# 2. 微博正文
# 3. 评论数
# 4. 转发数 
# 5. 点赞数
# 6. 发布时间
##################################################

"""
## 查 id
def findid(item: element.ResultSet) -> str:
    try:
        hrefstr = item.find('div', class_='avator').a.get("href")
        person_url = hrefstr.split("?")[0]
        person_id = person_url.split("/")[-1]
        return person_id
    except Exception as e:
        print (e)
        return ""

## 查微博正文
def find_content(item: element.ResultSet) -> str:
    try:
        contentstr = item.find('div', class_="content").p.text 
        # print (contentstr)
        return contentstr
    except Exception as e:
        print (e)
        return ""

def find_str_num(text : str) -> str:
    try:
        num = re.findall(r'\d+', text)
        return num[0]
    except Exception as e:
        return '0'

## 查评论数
def find_discuss_count(item: element.ResultSet) -> [str]:
    try:
        all_lis = item.find('div', class_="card-act").find_all("li")
        # 转发，评论， 点赞
        # print (all_lis[1].text, ", ", all_lis[2].text, ", ", all_lis[3].text)
        forward, discuss, thumbsUp =  find_str_num(all_lis[1].text), find_str_num(all_lis[2].text), find_str_num(all_lis[3].text)
        return [forward, discuss, thumbsUp]
    except Exception as e:
        print (e)
    return ['0', '0', '0']


## 查发布时间
def find_public_time(item: element.ResultSet) -> str:
    publish_time = ""
    try:
        publish_time =  item.find('p', class_="from").a.text
    except:
        return ""
    if "刚刚" in publish_time:
        publish_time = datetime.now().strftime('%Y-%m-%d %H:%M')
    elif "分钟" in publish_time:
        minute = publish_time[:publish_time.find("分钟")]
        minute = timedelta(minutes=int(minute))
        publish_time = (
            datetime.now() - minute).strftime(
            "%Y-%m-%d %H:%M")
    elif "今天" in publish_time:
        today = datetime.now().strftime("%Y-%m-%d")
        time = publish_time.replace('今天','')
        publish_time = today + " " + time

    elif "月" in publish_time:
        year = datetime.now().strftime("%Y")
        publish_time = str(publish_time)
        publish_time = year + "-" +publish_time.replace('月','-').replace('日','')
    else:
        publish_time = publish_time[:16]
    return publish_time


def ParseHTML(text :str) -> [{}]:
    # 假设html是需要被解析的html
    #将html传入BeautifulSoup 的构造方法,得到一个文档的对象
    soup = BeautifulSoup(text,'html.parser',from_encoding='utf-8')
    #查找所有的h4标签 
    all_card_wrap = soup.find_all('div',class_='card-wrap')
    all_item = []
    for item in all_card_wrap:
        weibo_id = findid(item)
        if weibo_id == "":
            continue
        weibo_content = find_content(item)
        forward, discuss, thumbsUp = find_discuss_count(item)
        publish_time = find_public_time(item)

        itemObj = {}
        itemObj["id"] = weibo_id
        itemObj["content"] = weibo_content
        itemObj["discuss"] = discuss
        itemObj["forward"] = forward
        itemObj["thumbsUp"] = thumbsUp
        itemObj["publish_time"] = publish_time 
        all_item.append(itemObj)
    return all_item

导出excel

加个导出excel功能

#!/usr/bin/env python3
# encoding:utf-8


import xlwt


def set_style(self,name,height,bold=False):
    style = xlwt.XFStyle() # 初始化样式
    font = xlwt.Font() # 为样式创建字体
    font.name = name # 'Times New Roman'
    font.bold = bold
    font.color_index = 4
    font.height = height
    borders= xlwt.Borders()
    borders.left= 6
    borders.right= 6
    borders.top= 6
    borders.bottom= 6
    style.font = font
    style.borders = borders
    return style

def write_to_excel(dataset,save_excel_name):
    head_row = ["微博id", "微博正文","评论数", "转发数", "点赞数", "发布时间"]
    head_key = ["id", "content", "discuss","forward", "thumbsUp", "publish_time"]
    f = xlwt.Workbook(encoding='utf-8') # 创建工作簿
    # 创建第一个sheet:
    count=1
    sheet1 = f.add_sheet(u'sheet1', cell_overwrite_ok=True) # 创建sheet
    # 首行标题：
    for p in range(len(head_row)):
        sheet1.write(0,p,head_row[p]) #
    for datamap in dataset:
        for key in head_key:
            pp = head_key.index(key)
            sheet1.write(count,pp, str(datamap.get(key))) 
        count+=1
    #print ("save_execl_name", save_excel_name)
    try:
        f.save(save_excel_name) # 保存文件
    except Exception as e:
        print ("save f error:",e)
